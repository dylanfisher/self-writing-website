    function handleScroll() {
        const windowBottom = window.innerHeight + window.scrollY;
        
        decayWords.forEach((word, index) => {
            const rect = word.getBoundingClientRect();
            const wordBottom = rect.top + rect.height + window.scrollY;
            
            if (wordBottom < 0) {
                if (!word.classList.contains('decayed')) {
                    word.classList.add('decayed');
                    word.style.transform = `translateY(${Math.random() * 10 + 2}px) rotate(${word.dataset.rotation})`;
                }
            } else if (wordBottom > windowBottom) {
                if (word.classList.contains('decayed')) {
                    word.classList.remove('decayed');
                    word.style.transform = 'none';
                }
            }
        });
    }

    window.addEventListener('scroll', handleScroll);

    function spawnTarget() {
        const area = document.getElementById('game-area');
        if (area.children.length > 15) return;

        const target = document.createElement('div');
        target.className = 'target';
        
        const maxX = area.clientWidth - 50;
        const maxY = area.clientHeight - 20;
        target.style.left = Math.floor(Math.random() * maxX) + 'px';
        target.style.top = Math.floor(Math.random() * maxY) + 'px';

        let dx = (Math.random() > 0.5 ? 1.5 : -1.5);
        let dy = (Math.random() > 0.5 ? 0.5 : -0.5);

        function moveTarget() {
            let l = parseFloat(target.style.left);
            let t = parseFloat(target.style.top);
            
            if (l <= 0 || l >= maxX) dx *= -1;
            if (t <= 0 || t >= maxY) dy *= -1;

            target.style.left = l + dx + 'px';
            target.style.top = t + dy + 'px';
            
            if(area.contains(target)) requestAnimationFrame(moveTarget);
        }

        moveTarget();
        
        target.addEventListener('click', (e) => {
            e.stopPropagation();
            
            if (surpriseMode) {
                // Evasion behavior
                target.style.display = 'none';
                triggerSurprise();
            } else {
                score += 1;
                surpriseTrigger += 1;
                
                // TRIGGER: After 10 "clicks" on normal targets, start the surprise
                if (surpriseTrigger >= 10) {
                    surpriseMode = true;
                    document.getElementById('arena-instruction').style.display = 'none';
                    document.getElementById('arena-msg').style.opacity = '1';
                }
                
                document.getElementById('score').innerText = String(score).padStart(4, '0');
                
                const ripple = document.createElement('div');
                ripple.style.position = 'absolute';
                ripple.style.left = (e.clientX - area.getBoundingClientRect().left) + 'px';
                ripple.style.top = (e.clientY - area.getBoundingClientRect().top) + 'px';
                ripple.style.width = '20px';
                ripple.style.height = '20px';
                ripple.style.border = '2px solid #ff9800';
                ripple.style.borderRadius = '50%';
                ripple.style.transform = 'translate(-50%, -50%) scale(0)';
                ripple.style.transition = 'transform 0.5s, opacity 0.5s';
                ripple.style.opacity = '1';
                ripple.style.pointerEvents = 'none';
                area.appendChild(ripple);

                requestAnimationFrame(() => {
                    ripple.style.transform = 'translate(-50%, -50%) scale(3)';
                    ripple.style.opacity = '0';
                });

                setTimeout(() => ripple.remove(), 500);
                target.remove();
            }
        });
        
        area.appendChild(target);
        setTimeout(() => {
            if (area.contains(target)) target.remove();
        }, 3000);
    }
